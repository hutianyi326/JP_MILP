"""One-time, hash-audited Japan package relocation; no overwrite or recursive delete."""
from pathlib import Path
import hashlib, json, os, re, ast, csv, io

ROOT = Path(__file__).resolve().parents[1]
JP = ROOT / 'countries/JP'
AUDIT = JP / 'organization_20260930'
assert not AUDIT.exists(), 'Migration already started; inspect the existing audit first.'

def sha(data):
    return hashlib.sha256(data).hexdigest()

def rel(p):
    return p.relative_to(ROOT).as_posix()

def safe(p):
    p = p.resolve()
    assert p.is_relative_to(ROOT), str(p)
    return p

mapping = {}
for src, dst in [('project/japan_research', 'countries/JP/research'),
                 ('data/raw/JP', 'countries/JP/data/raw')]:
    for p in (ROOT/src).rglob('*'):
        if p.is_file():
            mapping[rel(p)] = (Path(dst)/p.relative_to(ROOT/src)).as_posix()
for p in list(JP.glob('*.md')):
    mapping[rel(p)] = 'countries/JP/topics/' + p.name
mapping['model/JP_milp_formulation.md'] = 'countries/JP/model/JP_milp_formulation.md'
mapping['project/japan_research_scope_draft_20260928.md'] = 'countries/JP/research_scope_draft_20260928.md'

def immutable(old):
    return (old.startswith('data/raw/JP/') or any(old.startswith('project/japan_research/'+s+'/')
            for s in ['evidence','extracted','review_initial','review_round1','milp_review_initial'])
            or (old.startswith('project/japan_research/') and old.endswith('.json')))

for old, new in mapping.items():
    assert safe(ROOT/old).is_file(), old
    assert not safe(ROOT/new).exists(), new
assert len(set(mapping.values())) == len(mapping)
AUDIT.mkdir(parents=True)
records = []
for old, new in sorted(mapping.items()):
    p = ROOT/old
    data = p.read_bytes()
    records.append(dict(old_path=old, new_path=new, bytes_before=len(data), sha256_before=sha(data), immutable=immutable(old)))
(AUDIT/'migration_manifest.json').write_text(json.dumps({'status':'planned','files':records},ensure_ascii=False,indent=2),encoding='utf-8')

# Each source/destination was resolved and checked under this workspace before moving.
for old, new in mapping.items():
    dst = safe(ROOT/new)
    dst.parent.mkdir(parents=True,exist_ok=True)
    safe(ROOT/old).rename(dst)
for src in ['project/japan_research','data/raw/JP']:
    base = safe(ROOT/src)
    for d in sorted((p for p in base.rglob('*') if p.is_dir()),key=lambda p:len(p.parts),reverse=True):
        safe(d).rmdir()  # only succeeds for empty directories
    base.rmdir()

prefixes = [('project/japan_research','countries/JP/research'),
            ('data/raw/JP','countries/JP/data/raw')]
def map_path(old):
    if old in mapping:
        return mapping[old]
    for a,b in prefixes:
        if old == a or old.startswith(a+'/'):
            return b+old[len(a):]
    return old

def replace_root_paths(text):
    # Exact file replacements first, then directory references. Use placeholders
    # so a newly written destination cannot be replaced a second time.
    changes = []
    pairs = [(a,b) for a,b in mapping.items() if not a.startswith(('project/japan_research/','data/raw/JP/'))]
    pairs += prefixes
    for i,(a,b) in enumerate(sorted(pairs,key=lambda x:-len(x[0]))):
        for aa,bb in [(a,b),(a.replace('/','\\'),b.replace('/','\\'))]:
            token = f'__JP_MOVE_{len(changes)}__'
            if aa in text:
                text=text.replace(aa,token)
                changes.append((token,bb))
    for token,b in changes:
        text=text.replace(token,b)
    return text

def resolve_old_link(old, target):
    target=target.replace('\\','/')
    p=safe(ROOT/old).parent/target
    normalized=os.path.normpath(str(p))
    try:
        result=Path(normalized).relative_to(ROOT).as_posix()
    except ValueError:
        return None
    if result in mapping or (ROOT/result).exists():
        return result
    # Repair pre-existing links which incorrectly used repository-root paths
    # from a nested country folder.
    for marker in ['project/japan_research','data/raw/JP','countries/JP','model/','src/','project/']:
        pos=target.find(marker)
        if pos>=0:
            candidate=target[pos:]
            if candidate in mapping or (ROOT/candidate).exists():
                return candidate
    return result

link_re = re.compile(r'(!?\[[^\]\n]*\]\()([^\n]*?)(\))')
def update_text(old,new,text):
    held=[]
    if old.endswith('.md'):
        def link(m):
            target=m[2]
            if re.match(r'^(?:[a-zA-Z]+:|#)',target):
                return m[0]
            bare=target.strip('<>')
            path,sep,fragment=bare.partition('#')
            if not path:
                return m[0]
            resolved=resolve_old_link(old,path)
            if resolved is None:
                return m[0]
            relocated=map_path(resolved)
            dest=ROOT/relocated
            out=os.path.relpath(dest,(ROOT/new).parent).replace('\\','/')
            out += (sep+fragment)
            token=f'__JP_LINK_{len(held)}__'
            held.append((token,m[1]+out+m[3]))
            return token
        text=link_re.sub(link,text)
    # Within the relocated research directory, these references have a new depth.
    if old.startswith('project/japan_research/'):
        text=text.replace('../../data/raw/JP','../data/raw')
        text=text.replace('../../countries/JP/','../topics/')
        text=text.replace('../../model/JP_milp_formulation.md','../model/JP_milp_formulation.md')
    text=replace_root_paths(text)
    for token,value in held:
        text=text.replace(token,value)
    return text

changed=[]
for r in records:
    old,new=r['old_path'],r['new_path']
    if r['immutable'] or Path(old).suffix not in {'.md','.py','.csv','.jsonl'}:
        continue
    p=ROOT/new
    original=p.read_bytes()
    try:
        text=original.decode('utf-8-sig')
    except UnicodeDecodeError:
        continue
    updated=update_text(old,new,text)
    if updated != text:
        snapshot=AUDIT/'before_path_edits'/old
        snapshot.parent.mkdir(parents=True,exist_ok=True)
        snapshot.write_bytes(original)
        bom=b'\xef\xbb\xbf' if original.startswith(b'\xef\xbb\xbf') else b''
        p.write_bytes(bom+updated.encode('utf-8'))
        changed.append(new)

    if updated!=text:
        snap=AUDIT/'before_path_edits'/old
        snap.parent.mkdir(parents=True,exist_ok=True)
        snap.write_bytes(original)
        p.write_text(updated,encoding='utf-8',newline='')
        external.append(old)
